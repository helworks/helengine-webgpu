cmake_minimum_required(VERSION 3.26)
project(helengine_webgpu LANGUAGES CXX)
if(NOT EMSCRIPTEN)
    message(FATAL_ERROR "This target requires the Emscripten toolchain.")
endif()
set(HELENGINE_CORE_CPP_ROOT "" CACHE PATH "Official generated engine root")
set(HELENGINE_CODE_ROOT "" CACHE PATH "Staged generated gameplay root")
set(HELENGINE_CONTENT_ROOT "" CACHE PATH "Prepared content tree")
if(NOT EXISTS "${HELENGINE_CORE_CPP_ROOT}/helcpp_config.hpp")
    message(FATAL_ERROR "Generated engine sources are required.")
endif()
if(EXISTS "${HELENGINE_CORE_CPP_ROOT}/generated_windows_handoff.cmake")
    include("${HELENGINE_CORE_CPP_ROOT}/generated_windows_handoff.cmake")
endif()
if(EXISTS "${HELENGINE_CORE_CPP_ROOT}/helengine_core_unity.cpp")
    set(CORE_UNIT "${HELENGINE_CORE_CPP_ROOT}/helengine_core_unity.cpp")
elseif(EXISTS "${HELENGINE_CORE_CPP_ROOT}/helengine_core_amalgamated.cpp")
    set(CORE_UNIT "${HELENGINE_CORE_CPP_ROOT}/helengine_core_amalgamated.cpp")
elseif(DEFINED CPP_GENERATED_UNITY_SOURCE)
    set(CORE_UNIT "${CPP_GENERATED_UNITY_SOURCE}")
else()
    message(FATAL_ERROR "Official generated core translation unit is missing.")
endif()
file(GLOB_RECURSE HOST_SOURCES CONFIGURE_DEPENDS "src/*.cpp")
set(CORE_RUNTIME_SOURCES
    "${HELENGINE_CORE_CPP_ROOT}/runtime/runtime_startup_manifest.cpp"
    "${HELENGINE_CORE_CPP_ROOT}/runtime/runtime_scene_catalog_manifest.cpp"
    "${HELENGINE_CORE_CPP_ROOT}/runtime/runtime_code_module_manifest.cpp")
add_executable(helengine_webgpu ${HOST_SOURCES} ${CORE_UNIT} ${CORE_RUNTIME_SOURCES})
target_compile_features(helengine_webgpu PRIVATE cxx_std_20)
target_include_directories(helengine_webgpu PRIVATE src "${HELENGINE_CORE_CPP_ROOT}")
target_compile_options(helengine_webgpu PRIVATE --use-port=emdawnwebgpu -fexceptions -frtti)
target_link_options(helengine_webgpu PRIVATE --use-port=emdawnwebgpu -fexceptions -sALLOW_MEMORY_GROWTH=1 -sFORCE_FILESYSTEM=1 -sMODULARIZE=1 -sEXPORT_NAME=createHelEngine -sENVIRONMENT=web --preload-file "${HELENGINE_CONTENT_ROOT}@/content")
set_target_properties(helengine_webgpu PROPERTIES SUFFIX ".js")
if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_link_options(helengine_webgpu PRIVATE -O2 --closure=1)
endif()
